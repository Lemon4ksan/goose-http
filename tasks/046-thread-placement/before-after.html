Stopping a lane: goose-http

goose-http has no way to stop a lane today. examples/later never stops its lane, and the process ends around it. A program that does stop one sends a message, and a full box can refuse it.

Nowexamples/later/later.goose
thread_fn backend(box: http::Box) {
    var reply: u8[>..] = [];
    // Never ends: main's exit takes the thread with it, mid-job if need be.
    loop {
        http::mail_wait(box, -1);
        http::mail_take(box) { key, tag, bytes =>
            var j = Job { .. };
            if tag == JOB && http::mail_value(bytes, j) && http::mail_live(j.reply) {
                reply.clear();
                format(reply, primes_below(j.n), "\n");
                http::mail_send(j.reply, 0, reply);
            }
        };
    }
}

fn main() {
    // ...
    let b, err = http::mail_open(256, JOB_SIZE);
    http::exit_on_error(err);
    thread_spawn(backend, b);
    var ids: i64[>..] = [];
    for i in http::worker_count(cfg) { ids.push(thread_spawn(worker, cfg, http::address(b, 0))); }
    for id in ids { thread_wait(id); }
}
Afterexamples/later/later.goose
thread_fn backend(box: http::Box) {
    var reply: u8[>..] = [];
    // Runs until main stops the box and every job queued before is taken.
    while !http::mail_stopped(box) {
        // No time limit: mail_stop wakes this wait too.
        http::mail_wait(box, -1);
        http::mail_take(box) { key, tag, bytes =>
            var j = Job { .. };
            if tag == JOB && http::mail_value(bytes, j) && http::mail_live(j.reply) {
                reply.clear();
                format(reply, primes_below(j.n), "\n");
                http::mail_send(j.reply, 0, reply);
            }
        };
    }
    http::mail_close(box);
}

fn main() {
    // ...
    let b, err = http::mail_open(256, JOB_SIZE);
    http::exit_on_error(err);
    let lane = thread_spawn(backend, b);
    var ids: i64[>..] = [];
    for i in http::worker_count(cfg) { ids.push(thread_spawn(worker, cfg, http::address(b, 0))); }
    for id in ids { thread_wait(id); }
    // A flag in the box, not a message in a slot: a full box can't refuse it.
    http::mail_stop(b);
    thread_wait(lane);
}
New: http::mail_stop(b: Box) http::mail_stopped(b: Box) -> bool · Unchanged: mail_wait, whose false still means the time limit passed

Stopping a lane: goose-rest

Every lane in the tutorial and the best practices writes the same loop by hand, with a STOP tag and a running flag. rest::lane replaces both.

Nowbest-practices/lanes/main.goose
let STOP: i32 = 5;   // in app.goose, beside the job tags

thread_fn hasher(box: http::Box) {
    var running = true;
    while running {
        http::mail_wait(box, 1000);
        http::mail_take(box) { key, tag, bytes =>
            if tag == VERIFY_PASSWORD {
                let a, ok = rest::asked<auth::SignInJob>(bytes);
                if ok { rest::reply_value(a, auth::verify_password(a.job)); }
            } else if tag == STOP {
                running = false;
            }
        };
    }
    http::mail_close(box);
}

fn main() {
    // ...
    thread_wait(thread_spawn(worker, cfg, app));
    // FULL if the box is full, and nothing reads the result:
    // the lane then never stops, and thread_wait hangs.
    http::mail_send(app.hasher, STOP, "");
    thread_wait(ht);
}
Afterbest-practices/lanes/main.goose
// No STOP tag: a stop isn't a job.

thread_fn hasher(box: http::Box) {
    // Each job in turn until the box is stopped and empty; then closes it.
    rest::lane(box) { tag, bytes =>
        if tag == VERIFY_PASSWORD {
            let a, ok = rest::asked<auth::SignInJob>(bytes);
            if ok { rest::reply_value(a, auth::verify_password(a.job)); }
        }
    };
}

fn main() {
    // ...
    thread_wait(thread_spawn(worker, cfg, app));
    http::mail_stop(hbox);
    thread_wait(ht);
}

// In goose-rest: the loop each lane wrote by hand.
fn lane(box: http::Box, body) {
    while !http::mail_stopped(box) {
        http::mail_wait(box, -1);
        http::mail_take(box) { key, tag, bytes => body(tag, bytes); };
    }
    http::mail_close(box);
}
New: rest::lane(box) { tag, bytes => } · Gone from apps: STOP var running mail_wait(box, 1000)

Placement

Tina names one core per shard in its boot spec (target_core). Here any thread, a worker or a lane, takes a placement: floating (today, and the default), one core each from a set, one named core, or anywhere within a set. A set is what lets a program plan around SMT siblings, L3 groups, and NUMA nodes. Only placements that beat floating on a short route's p99 by more than the runs' spread stay.

Nowa program with a CPU lane
thread_fn hasher(box: http::Box) {
    // Runs on whichever core the OS picks, at the workers' priority:
    // a 50 ms hash competes with them as an equal.
    rest::lane(box) { tag, bytes =>
        // ...
    };
}

fn main() {
    // ...
    let ht = thread_spawn(hasher, hbox);
    // workers: 0 starts one per hardware thread, none of them pinned.
    let cfg = http::Config { port: 8080, mail_slots: 64, max_waiting: 1024, .. };
    // ...
}
After8 cores, 16 hardware threads (Ryzen 7 5800X3D: CPU n and n+8 share a core)
// A placement is a flat value, so a thread can take it as an argument.
//   http::floating()        the OS decides (the default)
//   http::each_core(set)    each thread pinned to its own core of the set,
//                           physical cores first, SMT siblings last
//   http::core(n)           pinned to hardware thread n
//   http::within(set)       the OS moves it, but only within the set
// A set is written as taskset and cgroups write it: "0-5,8".

fn main() {
    // Six workers, each alone on a physical core.
    let cfg = http::Config { port: 8080, workers: 6,
                             place: http::each_core("0-5"), .. };

    // The hasher has a core of its own; the writer has the other.
    let hasher = rest::lane_open(1, HASHER_SLOTS, 256, http::core(6));
    let writer = rest::lane_open(1, WRITER_SLOTS, 256, http::core(7));
    // Four readers on the workers' SMT siblings: they mostly wait on
    // SQLite, so they take little from the worker sharing their core.
    let reports = rest::lane_open(4, REPORTS_SLOTS, 256, http::within("8-15"));
    // ...
}

thread_fn reader(box: http::Box, place: http::Place) {
    // First, before the thread touches its own memory: Linux puts a page
    // on the NUMA node of the thread that first writes it.
    http::place_thread(place);
    rest::lane(box) { tag, bytes => /* ... */ };
}

// On a two-node server the same program plans by node:
//   workers  http::each_core(http::node_cores(0))
//   readers  http::within(http::node_cores(1)), beside the database's disk

// For any machine, 1 core included: nothing pinned, and the kernel runs
// a worker ahead of the hasher whenever both are ready.
thread_fn hasher(box: http::Box) {
    http::low_priority();   // nice +10 and SCHED_BATCH on Linux;
                            // a background QoS class on macOS
    rest::lane(box) { tag, bytes => /* ... */ };
}

// macOS has no affinity: every placement there reports that it did
// nothing, and workers keep the performance-core QoS class they get now.

A lane of several threads

With a real database (SQLite or Turso), splitting data by key barely matters. What remains is one concept: a lane is N threads, each with its own box and its own database connection. A job goes to the first thread with room, or, when a key is given, to the thread that key always picks. The writer is a lane of 1.

Nowtutorial: "more threads, each with its own box"
// The app opens each thread's box and keeps every address.
struct App { reports: http::MailAddress[..4] }

fn main() {
    var app = App { reports: [..4] };
    for i in 4 {
        let b, e = http::mail_open(REPORTS_SLOTS, 256);
        http::exit_on_error(e);
        app.reports.push(http::address(b, 0));
        // ... spawn a reports thread on b, keep its id ...
    }
}

fn project(cx) {
    // ...
    // The route picks a thread itself, and FULL on that one is a 503
    // even when another thread is idle.
    let to = cx.app.reports[project % cx.app.reports.len];
    rest::ask(cx, to, TOTAL_ENTRIES, job, Waiting.ReportTotals);
}
Aftersketch
// A lane: N boxes of the same size, and their addresses.
struct App { writer: rest::Lane, reports: rest::Lane, hasher: rest::Lane }

fn main() {
    // One writer: SQLite allows one. Four readers, each with its own
    // read connection. Two hashers.
    let writer = rest::lane_open(1, WRITER_SLOTS, 256);
    let reports = rest::lane_open(4, REPORTS_SLOTS, 256);
    let hasher = rest::lane_open(2, HASHER_SLOTS, 256);
    // ... spawn one thread per box: writer.boxes[i], and so on ...
}

fn project(cx) {
    // ...
    // Any reader: the first whose box has room. 503 only when all are full.
    rest::ask(cx, cx.app.reports, TOTAL_ENTRIES, job, Waiting.ReportTotals);
}

fn start(cx) {
    // ...
    // By key: one member's timer jobs always reach the same thread, so
    // they run in the order they were sent.
    rest::ask_by(cx, cx.app.writer, member, START_TIMER, job, Waiting.TimerStart);
}

Group commit on the writer

The writer takes every job waiting in its box, runs each in a savepoint of its own, and commits once. Then it sends the replies, so no route hears "created" before the data is durable. One fsync covers the whole group, which is the largest win on writes that SQLite allows. Goose has no SQLite binding yet, so db:: is hypothetical here.

Nowthe writer, one transaction per job
thread_fn writer(box: http::Box) {
    let conn = db::open("app.db");
    rest::lane(box) { tag, bytes =>
        if tag == CREATE_ENTRY {
            let a, ok = rest::asked<entries::CreateJob>(bytes);
            if ok {
                // BEGIN ... COMMIT inside the job: an fsync per entry.
                rest::reply_value(a, entries::create_entry(conn, a.job));
            }
        }
    };
}
Aftersketch
thread_fn writer(box: http::Box) {
    let conn = db::open("app.db");
    // Each wake: BEGIN, every job in the box, COMMIT, then the replies.
    rest::lane_batch(box, conn) { tag, bytes =>
        if tag == CREATE_ENTRY {
            let a, ok = rest::asked<entries::CreateJob>(bytes);
            if ok {
                // In a savepoint of its own: a job that fails rolls back
                // alone, and the others still commit. reply_value holds
                // the reply until the COMMIT returns.
                rest::reply_value(a, entries::create_entry(conn, a.job));
            }
        }
    };
}
// The job stays one use case and one unit of work; only the commit
// is shared. A job's function doesn't change.

Crashesdocs only
// A crash in any thread, an index out of bounds in a lane's job say,
// ends the whole process: Goose can't catch an abort (spec §9.3).
// The service manager restarts it, after a delay, and gives up after
// a few crashes in a row:
//
//   systemd     Restart=on-failure, RestartSec=1,
//               StartLimitIntervalSec=60, StartLimitBurst=5
//   Compose     restart: on-failure:5
//   Kubernetes  restarts by default; CrashLoopBackOff spaces them
//
// No supervisor in the package, and no lanes as processes. A restart
// per thread waits on Goose (goose-findings: "An abort ends every
// thread of the program").
Routes that waitunchanged
// Already Tina's model: ctx_call with a timeout, then WAIT_REPLY.
enum Waiting {
    None,
    EntryCreate,               // entries::create
    ReportTotals,              // reports::project
    AuthPassword, AuthSession, // auth::sign_in
}

fn create(cx) {
    // ...
    // FULL is .mailbox_full: 503 at once.
    rest::ask(cx, cx.app.writer, CREATE_ENTRY, CreateJob { input: input },
              Waiting.EntryCreate);
}

// The reply's case. A late reply is TAG_CALL_TIMEOUT: 504.
fn resume(w: Waiting.EntryCreate, cx) {
    let r, ok = rest::read_reply<CreateReply>(cx);
    guard ok else { rest::fail(cx, 502, "bad_reply"); }
    check(r.err, cx);
    rest::json(cx, 201, r.entry);
}
